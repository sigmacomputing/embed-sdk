```tsx
import { useSigmaIframe } from "@sigmacomputing/react-embed-sdk";
function BasicExample({ src }: { src: string }) {
  const { iframeRef, loading, error } = useSigmaIframe(); // [!code highlight]
  return (
    <>
      {loading && <p>Loading...</p>} // [!code highlight]
      {error && <p>Error loading iframe</p>} // [!code highlight]
      <iframe
        src={src}
        ref={iframeRef} // [!code highlight]
        className={loading || error ? "opacity-0 pointer-events-none" : ""} // [!code highlight]
        width={"100%"}
        height={"100%"}
      />
    </>
  );
}
```
